from collections import Counter

BASELINE_DAYS = 30

MIN_BASELINE_EVENTS = 20
MEDIUM_BASELINE_EVENTS = 50

# Adaptive UEBA Learning Guard

def _circular_hour_difference(hour1, hour2):
    difference = abs(hour1 - hour2)
    return min(difference, 24 - difference)


def _get_confidence(event_count):
    if event_count < MIN_BASELINE_EVENTS:
        return "LEARNING", 0.25

    if event_count < MEDIUM_BASELINE_EVENTS:
        return "LOW", 0.50

    if event_count < HIGH_BASELINE_EVENTS:
        return "MEDIUM", 0.75

    return "HIGH", 1.0


def _get_learning_events(
    db,
    user,
    since,
    reference_time
):
    """
    Adaptive UEBA Learning Guard

    媛쒖씤 ?됰룞 湲곗??좎뿉??LOW ?꾪뿕?꾨줈
    理쒖쥌 ?먯젙???대깽?몃쭔 ?ъ슜?쒕떎.

    MEDIUM / HIGH / CRITICAL ?대깽?몃뒗
    蹂댁븞 ?대깽??湲곕줉?먮뒗 ?⑥븘 ?덉?留?
    媛쒖씤 ?뺤긽 ?됰룞 ?숈뒿?먯꽌???쒖쇅?쒕떎.
    """

    learning_since = since

    monitored_user = (
        db.query(MonitoredUser)
        .filter(MonitoredUser.username == user)
        .first()
    )

    # ?ㅼ젣 紐⑤땲?곕쭅 ??곸쑝濡??깅줉???ъ슜?먮뒗
    # ?깅줉 ?쒖젏 ?댄썑???됰룞留?媛쒖씤 湲곗????숈뒿???ъ슜?쒕떎.

    # Adaptive UEBA ?쒖뿰 ?ъ슜?먮뒗 湲곗〈 30??湲곗???洹몃?濡??ъ슜?쒕떎.
    if (
        monitored_user is not None
        and monitored_user.created_at is not None
        and monitored_user.created_at > learning_since
    ):
        learning_since = monitored_user.created_at

    events = (
        db.query(SecurityEvent)
        .filter(
            SecurityEvent.user == user,
            SecurityEvent.timestamp >= learning_since,
            SecurityEvent.timestamp < reference_time,
            SecurityEvent.risk_level == LEARNING_RISK_LEVEL
        )
        .order_by(SecurityEvent.timestamp.asc())
        .all()
    )

    return [
        event
        for event in events
        if is_event_allowed_for_baseline(db, event)
    ]


def _get_active_hours(events):
    hour_counter = Counter()

    for event in events:
        if event.timestamp:
            hour_counter[event.timestamp.hour] += 1

    if not hour_counter:
        return []

    maximum_count = max(hour_counter.values())

    threshold = max(
        2,
        int(maximum_count * 0.30)
    )

    active_hours = [
        hour
        for hour, count in hour_counter.items()
        if count >= threshold
    ]

    return sorted(active_hours)


def _get_active_weekdays(events):
    weekday_counter = Counter()

    for event in events:
        if event.timestamp:
            weekday_counter[event.timestamp.weekday()] += 1

    if not weekday_counter:
        return []

    maximum_count = max(weekday_counter.values())

    threshold = max(
        2,
        int(maximum_count * 0.25)
    )

    active_weekdays = [
        weekday
        for weekday, count in weekday_counter.items()
        if count >= threshold
    ]

    return sorted(active_weekdays)


def _get_known_ips(events):
    ip_counter = Counter()

    for event in events:
        if event.source_ip:
            ip_counter[event.source_ip] += 1

    return {
        ip
        for ip, count in ip_counter.items()
        if count >= 2
    }


def _get_normal_processes(events):
    process_counter = Counter()

    for event in events:
        if (
            event.event_type == "process_execution"
            and event.process_name
        ):
            process_name = event.process_name.lower()
            process_counter[process_name] += 1

    return {
        process
        for process, count in process_counter.items()
        if count >= 2
    }


def _get_average_file_access(events):
    file_counts = [
        event.file_access_count
        for event in events
        if (
            event.event_type == "file_access"
            and event.file_access_count is not None
            and event.file_access_count > 0
        )
    ]

    if not file_counts:
        return 0

    return sum(file_counts) / len(file_counts)


def _get_usb_usage_rate(events):
    if not events:
        return 0

    usb_events = [
        event
        for event in events
        if (
            event.event_type == "usb_connected"
            or event.usb_connected == 1
        )
    ]

    return len(usb_events) / len(events)

def _get_peer_users(
    db,
    user
):
    """
    媛숈? department + role???쒖꽦 紐⑤땲?곕쭅 ?ъ슜?먮? Peer Group?쇰줈 援ъ꽦?쒕떎.
    ?먭린 ?먯떊? ?쒖쇅?쒕떎.

    Peer媛 異⑸텇?섏? ?딆쑝硫??먯닔瑜?留뚮뱾吏 ?딅뒗??
    """
    target_user = (
        db.query(MonitoredUser)
        .filter(MonitoredUser.username == user)
        .first()
    )

    if target_user is None:
        return []

    if not target_user.department or not target_user.role:
        return []

    peer_rows = (
        db.query(MonitoredUser)
        .filter(
            MonitoredUser.username != user,
            MonitoredUser.department == target_user.department,
            MonitoredUser.role == target_user.role,
            MonitoredUser.monitoring_enabled == 1
        )
        .all()
    )

    return peer_rows


def _get_peer_learning_events(
    db,
    peer_users,
    since,
    reference_time
):
    """
    Peer Group??Learning Guard瑜??숈씪?섍쾶 ?곸슜?쒕떎.

    媛?Peer??愿由ъ옄 ?깅줉 ?쒖젏 ?댄썑??LOW ?대깽?몃쭔 ?ъ슜?쒕떎.
    ?곕씪??怨쇨굅 怨듦꺽 ?대깽?몃굹 ?깅줉 ?댁쟾 濡쒓렇媛 Peer ?뺤긽 湲곗??좎쓣
    ?ㅼ뿼?쒗궎吏 ?딅뒗??
    """
    events = []

    for peer in peer_users:
        peer_events = _get_learning_events(
            db=db,
            user=peer.username,
            since=since,
            reference_time=reference_time
        )

        events.extend(peer_events)

    events.sort(
        key=lambda event: (
            event.timestamp or datetime.min,
            event.id or 0
        )
    )

    return events

    """

        reference_time = datetime.now()

    since = reference_time - timedelta(
        days=BASELINE_DAYS
    )

    peer_users = _get_peer_users(
        db=db,
        user=user
    )

    peer_events = _get_peer_learning_events(
        db=db,
        peer_users=peer_users,
        since=since,
        reference_time=reference_time
    )

    target_user = (
        db.query(MonitoredUser)
        .filter(MonitoredUser.username == user)

            f"{target_user.department or '-'} / "

        len(peer_users) >= MIN_PEER_USERS
        and len(peer_events) >= MIN_PEER_EVENTS

        "user": user,
        "group_key": group_key,
        "available": available,
        "minimum_peer_users": MIN_PEER_USERS,

            peer_events
        ),
        "active_weekdays": _get_active_weekdays(
            peer_events
        ),
        "known_ips": sorted(
            _get_known_ips(peer_events)
        ),
        "average_file_access": round(
            _get_average_file_access(peer_events),
            2
        ),
        "normal_processes": sorted(
            _get_normal_processes(peer_events)
        ),
        "usb_usage_rate_percent": round(
            _get_usb_usage_rate(peer_events) * 100,

def _already_known_in_low_history(db, event, change_type, change_value):
    from server.models import SecurityEvent

    q = db.query(SecurityEvent).filter(
        SecurityEvent.user == event.user,
        SecurityEvent.id != event.id,
        SecurityEvent.risk_level == "LOW",
        SecurityEvent.timestamp < event.timestamp,
    )

    if change_type == "IP":
        q = q.filter(SecurityEvent.source_ip == change_value)
        return q.first() is not None

    if change_type == "PROCESS":
        rows = q.filter(SecurityEvent.process_name.isnot(None)).all()
        target = change_value.lower()
        return any(
            str(row.process_name).strip().lower() == target
            for row in rows
        )

    if change_type == "DEVICE":
        rows = q.all()
        for row in rows:
            rid = getattr(row, "device_id", None)
            rname = getattr(row, "device_name", None)
            if rid and str(rid).strip() == change_value:
                return True
            if (not rid) and rname and str(rname).strip() == change_value:
                return True
        return False

    if change_type == "HOST":
        if hasattr(SecurityEvent, "host"):
            q = q.filter(SecurityEvent.host == change_value)
            return q.first() is not None
        return False

    return False

                "stages"

        event

        _event_time(event)

        or 0
    )

    adaptive_reasons = (
        adaptive_result.get(
            "reasons",
            []
        )
        or []
    )

    reasons.extend(
        adaptive_reasons
    )

    compensation_time = (
        perf_counter()
        - step_start
    )

    # --------------------------------
    # 8. Rule + ML 湲곕낯 ?꾪뿕??
    # --------------------------------

    step_start = perf_counter()

    base_score, _ = calculate_final_risk(
        rule_score,
        anomaly_score
    )

    # --------------------------------
    # 9. Baseline + Sequence + Peer Group + Adaptive 蹂댁젙
    # --------------------------------

    final_score = (
        base_score
        + baseline_score * 0.3
        + sequence_score * 0.4
        + peer_contribution
        - adaptive_compensation
    )

    final_score = round(
        max(
            min(final_score, 100),
            0
        ),
        2
    )

    score_before_floor = final_score
    floor_applied = False
    floor_reason = None

    # --------------------------------
    # 10. Sequence 理쒖냼 ?꾪뿕??
    # --------------------------------

    if sequence_score >= 100:
        final_score = max(
            final_score,
            90
        )

    elif sequence_score >= 80:
        final_score = max(
            final_score,
            80
        )

    elif sequence_score >= 60:
        final_score = max(
            final_score,
            65
        )

    elif sequence_score >= 40:
        final_score = max(
            final_score,
            45
        )

    # --------------------------------
    # Baseline 理쒖냼 ?꾪뿕??
    # --------------------------------

    if baseline_score >= 60:
        final_score = max(
            final_score,
            60
        )

    elif baseline_score >= 30:
        final_score = max(
            final_score,
            30
        )

    if final_score > score_before_floor:
        floor_applied = True

        if sequence_score >= 100:
            floor_reason = "Sequence Score 100 ?댁긽 ??Event Risk 理쒖냼 90"
        elif sequence_score >= 80:
            floor_reason = "Sequence Score 80 ?댁긽 ??Event Risk 理쒖냼 80"
        elif sequence_score >= 60:
            floor_reason = "Sequence Score 60 ?댁긽 ??Event Risk 理쒖냼 65"
        elif baseline_score >= 60:
            floor_reason = "Personal Baseline Score 60 ?댁긽 ??Event Risk 理쒖냼 60"
        elif sequence_score >= 40:
            floor_reason = "Sequence Score 40 ?댁긽 ??Event Risk 理쒖냼 45"
        elif baseline_score >= 30:
            floor_reason = "Personal Baseline Score 30 ?댁긽 ??Event Risk 理쒖냼 30"

    # --------------------------------
    # 11. 理쒖쥌 ?꾪뿕 ?깃툒
    # --------------------------------

    risk_level = get_risk_level(
        final_score
    )

    calculation_time = perf_counter() - step_start

    # --------------------------------
    # 12. DB 理쒖쥌 ?낅뜲?댄듃
    # --------------------------------

    step_start = perf_counter()

    db_event.risk_score = final_score
    db_event.risk_level = risk_level
    db_event.reasons = ", ".join(
        reasons
    )

    db.commit()
    db.refresh(db_event)

    # --------------------------------
    # 12-1. Behavior Change Verification
    # --------------------------------
    # HOST / WORKSTATION 以묐났 寃利?諛⑹?
    db_event.workstation_name = getattr(
        event,
        "workstation_name",
        None
    )

    change_results = evaluate_behavior_change(
        db=db,
        event=db_event
    )

    if change_results:
        for change in change_results:
            reasons.append(
                "?됰룞 蹂??寃利? "
                f"{change['type']}={change['value']} / "
                f"{change['status']} / "
                f"{change['reason']}"
            )

        db_event.reasons = ", ".join(reasons)
        db.commit()
        db.refresh(db_event)


    # --------------------------------
    # ?뺤옣 ?됰룞 蹂??寃利?
    # --------------------------------
    extended_change_results = evaluate_extended_behavior_changes(
        db=db,
        event=event,
        risk_score=final_score,
        risk_level=risk_level,
        reasons=reasons,
        reference_time=now,
    )

    if extended_change_results:
        for change in extended_change_results:
            reasons.append(
                "?됰룞 蹂??寃利? "

            )

        db_event.reasons = ", ".join(reasons)

    existing_change_results = (

        if "change_results" in locals()
        else []
    )

    persist_event_context(
        db=db,
        db_event=db_event,
        event=event,
        change_results=(
            list(existing_change_results)
            + list(extended_change_results)
        ),
    )

    risk_explanation = build_risk_explanation(
        event_id=db_event.id,

        base_score=base_score,
        baseline_score=baseline_score,

    )

    save_risk_explanation(
        db=db,
        event_id=db_event.id,
        explanation=risk_explanation
    )

    db_update_time = perf_counter() - step_start

    # --------------------------------
    # 13. Dynamic User Risk
    # --------------------------------
    # ?꾩옱 ?대깽?몄쓽 Event Risk ??μ씠 ?앸궃 ???ъ슜???꾩껜 ?꾪뿕 ?곹깭瑜?怨꾩궛?쒕떎.
    # ??媛믪? Event Risk???ㅼ떆 ?뷀븯吏 ?딆븘 ?먭린利앺룺??諛⑹??쒕떎.

    dynamic_risk = calculate_dynamic_user_risk(
        db=db,

    )

    dynamic_risk_time = perf_counter() - step_start

    # --------------------------------
    # 14. Incident Correlation
    # --------------------------------
    # Event Risk / Dynamic User Risk 怨꾩궛??蹂寃쏀븯吏 ?딅뒗??
    # ??μ씠 ?꾨즺??SecurityEvent?ㅼ쓣 蹂꾨룄???ш퀬 ?⑥쐞濡??곴?遺꾩꽍?쒕떎.
    step_start = perf_counter()

    try:

    except Exception as error:

        }

    incident_time = perf_counter() - step_start
    total_time = perf_counter() - total_start

    if perf_logging:

@app.post("/analyze-document")
async def analyze_document_api(
    file: UploadFile = File(...)
):
    if not file.filename:
        raise HTTPException(
            status_code=400,
            detail="?뚯씪 ?대쫫???놁뒿?덈떎."
        )

    if not file.filename.lower().endswith(
        ".docx"
    ):
        raise HTTPException(
            status_code=400,
            detail="?꾩옱??DOCX ?뚯씪留?遺꾩꽍?????덉뒿?덈떎."
        )

    temp_path = None

    try:
        with tempfile.NamedTemporaryFile(
            delete=False,
            suffix=".docx"
        ) as temp_file:

            shutil.copyfileobj(
                file.file,
                temp_file
            )

            temp_path = temp_file.name

        result = analyze_document(
            temp_path
        )

        return {
            "file_name": file.filename,
            "pii_score": result[
                "pii_score"
            ],
            "file_grade": result[
                "file_grade"
            ],
            "reasons": result[
                "reasons"
            ],
            "detected": result[
                "detected"
            ]
        }

    except Exception as error:
        raise HTTPException(
            status_code=500,
            detail=(
                f"臾몄꽌 遺꾩꽍 ?ㅽ뙣: "
                f"{str(error)}"
            )
        )

    finally:
        await file.close()

        if (
            temp_path
            and os.path.exists(temp_path)
        ):
            os.remove(temp_path)

            )

            file_access_count = int(
                np.clip(
                    np.random.normal(5, 2),
                    0,
                    12
                )
            )

            usb_connected = np.random.choice(
                [0, 0, 0, 0, 0, 1]
            )

            # ?뺤긽 洹쇰Т?쒓컙 08~19?쒕? 以묒떖?쇰줈 ?앹꽦

                np.clip(
                    np.random.normal(13, 3),
                    8,
                    19
                )
            )

            normal_data.append([
                login_fail_count,
                file_access_count,
                usb_connected,
                hour
            ])

        normal_data = np.array(normal_data)

        self.model.fit(normal_data)
        self.is_trained = True

    def analyze(
        self,
        login_fail_count,
        file_access_count,
        usb_connected,
        hour
    ):
        features = np.array([[
            login_fail_count,
            file_access_count,
            usb_connected,
            hour
        ]])

        prediction = self.model.predict(features)[0]

        raw_score = self.model.decision_function(features)[0]

        anomaly_score = (0.15 - raw_score) / 0.30 * 100

def detect_pii(text):
    results = {
        "resident_number": [],
        "phone_number": [],
        "email": [],
        "account_number": []
    }

    resident_pattern = r"\b\d{6}-[1-4]\d{6}\b"

    phone_pattern = r"\b01[016789]-\d{3,4}-\d{4}\b"

    email_pattern = (
        r"\b[A-Za-z0-9._%+-]+"
        r"@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b"
    )

    account_pattern = r"\b\d{2,6}-\d{2,6}-\d{2,6}\b"

    results["resident_number"] = re.findall(
        resident_pattern,
        text
    )

    results["phone_number"] = re.findall(
        phone_pattern,
        text
    )

    results["email"] = re.findall(
        email_pattern,
        text
    )

    account_candidates = re.findall(
        account_pattern,
        text
    )

    results["account_number"] = [
        account
        for account in account_candidates
        if account not in results["phone_number"]
        and account not in results["resident_number"]
    ]

    return results

    account_count = len(results["account_number"])

        score += min(resident_count * 40, 80)

        score += min(phone_count * 10, 30)
        reasons.append(
            f"?꾪솕踰덊샇 {phone_count}嫄??먯?"
        )

    if email_count > 0:
        score += min(email_count * 5, 20)
        reasons.append(
            f"?대찓??{email_count}嫄??먯?"
        )

    if account_count > 0:
        score += min(account_count * 20, 40)
        reasons.append(
            f"怨꾩쥖踰덊샇 ?꾨낫 {account_count}嫄??먯?"

def _safe_json_load(value, default):
    try:
        parsed = json.loads(value or "")
        return parsed
    except (TypeError, json.JSONDecodeError):
        return default


def _serialize_execution(execution):
    steps = _safe_json_load(execution.steps_json, [])
    history = _safe_json_load(execution.history_json, [])

    completed_count = sum(
        1
        for step in steps
        if step.get("status") in {"COMPLETED", "SKIPPED"}
    )

    approved_count = sum(
        1
        for step in steps
        if step.get("status") == "APPROVED"
    )

    total_count = len(steps)

    progress_percent = (
        round((completed_count / total_count) * 100, 1)
        if total_count
        else 0.0
    )

    return {
        "id": execution.id,
        "incident_id": execution.incident_id,
        "playbook_key": execution.playbook_key,
        "playbook_name": execution.playbook_name,
        "status": execution.status,
        "steps": steps,
        "history": history,
        "completed_step_count": completed_count,
        "approved_step_count": approved_count,
        "total_step_count": total_count,
        "progress_percent": progress_percent,
        "created_at": execution.created_at,
        "updated_at": execution.updated_at
    }


def _append_history(history, action, step=None, note=None):
    history.append({
        "timestamp": _now_text(),
        "action": action,
        "step_id": (
            step.get("step_id")
            if step
            else None
        ),
        "step_title": (
            step.get("title")
            if step
            else None
        ),
        "note": note
    })


def _refresh_execution_status(execution, steps):
    if steps and all(
        step.get("status") in {"COMPLETED", "SKIPPED"}
        for step in steps
    ):
        execution.status = "COMPLETED"
        return

    containment_steps = [
        step
        for step in steps
        if step.get("containment")
    ]

    if containment_steps and any(
        step.get("status") == "COMPLETED"
        for step in containment_steps
    ):
        execution.status = "CONTAINED"
        return

    if any(
        step.get("status") in {
            "APPROVED",
            "COMPLETED",
            "SKIPPED"
        }
        for step in steps
    ):
        execution.status = "IN_PROGRESS"
        return

    execution.status = "PENDING"


def get_or_create_playbook(
    db: Session,
    incident_id: int
):
    incident = get_incident(
        db=db,
        incident_id=incident_id
    )

    if incident is None:
        return None, "蹂댁븞 ?ш퀬瑜?李얠쓣 ???놁뒿?덈떎."

    existing = (
        db.query(ResponsePlaybookExecution)
        .filter(
            ResponsePlaybookExecution.incident_id
            == incident_id
        )
        .first()
    )

    if existing:
        return _serialize_execution(existing), None

    playbook_key, playbook_name, steps = (
        _select_playbook(incident)
    )

    history = []
    _append_history(
        history,
        "PLAYBOOK_CREATED",
        note=(
            f"{incident.get('incident_type')} / "
            f"{incident.get('severity')} Incident???곕씪 "
            f"'{playbook_name}' Playbook???앹꽦?덉뒿?덈떎."
        )
    )

    execution = ResponsePlaybookExecution(
        incident_id=incident_id,
        playbook_key=playbook_key,
        playbook_name=playbook_name,
        status="PENDING",
        steps_json=json.dumps(
            steps,
            ensure_ascii=False
        ),
        history_json=json.dumps(
            history,
            ensure_ascii=False
        )
    )

    db.add(execution)
    db.commit()
    db.refresh(execution)

    return _serialize_execution(execution), None


def update_playbook_step(
    db: Session,
    incident_id: int,
    step_id: str,
    action: str,
    note=None
):
    execution = (
        db.query(ResponsePlaybookExecution)
        .filter(
            ResponsePlaybookExecution.incident_id
            == incident_id
        )
        .first()
    )

    if execution is None:
        _, error = get_or_create_playbook(
            db=db,
            incident_id=incident_id
        )

        if error:
            return None, error

        execution = (
            db.query(ResponsePlaybookExecution)
            .filter(
                ResponsePlaybookExecution.incident_id
                == incident_id
            )
            .first()
        )

    steps = _safe_json_load(
        execution.steps_json,
        []
    )

    history = _safe_json_load(
        execution.history_json,
        []
    )

    target = next(
        (
            step
            for step in steps
            if step.get("step_id") == step_id
        ),
        None
    )

    if target is None:
        return None, "Playbook ?④퀎瑜?李얠쓣 ???놁뒿?덈떎."

    normalized_action = str(action or "").upper()

    if normalized_action == "APPROVE":
        if not target.get("approval_required"):
            return None, "???④퀎??蹂꾨룄 ?뱀씤???꾩슂?섏? ?딆뒿?덈떎."

        if target.get("status") == "COMPLETED":
            return None, "?대? ?꾨즺???④퀎?낅땲??"

        target["status"] = "APPROVED"
        target["approved_at"] = _now_text()
        target["note"] = note

        _append_history(
            history,
            "STEP_APPROVED",
            step=target,
            note=note
        )

    elif normalized_action == "COMPLETE":
        if (
            target.get("approval_required")

def migrate_security_event_schema(engine):
    inspector = inspect(engine)

    if "security_events" not in inspector.get_table_names():
        return []

    existing_columns = {
        column["name"]
        for column in inspector.get_columns("security_events")
    }

    added_columns = []

    with engine.begin() as connection:
        for column_name, column_type in SECURITY_EVENT_COLUMNS.items():
            if column_name in existing_columns:
                continue

            connection.execute(
                text(
                    f"ALTER TABLE security_events "
                    f"ADD COLUMN {column_name} {column_type}"
                )
            )
            added_columns.append(column_name)

    return added_columns
